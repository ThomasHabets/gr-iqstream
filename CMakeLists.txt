cmake_minimum_required(VERSION 3.16.3)
project(gr-iqstream VERSION 0.1.0 LANGUAGES CXX)
# GNU Radio 3.10.12 uses the legacy FindPythonInterp/FindBoost modules.
foreach(compat_policy CMP0148 CMP0167)
    if(POLICY ${compat_policy})
        cmake_policy(SET ${compat_policy} OLD)
    endif()
endforeach()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Gnuradio 3.10.12 EXACT REQUIRED COMPONENTS runtime blocks)
find_package(Protobuf 3.15 REQUIRED)
find_package(gRPC 1.51 CONFIG REQUIRED)
find_package(Threads REQUIRED)
option(ENABLE_PYTHON "Build Python bindings and GRC definitions" ON)
option(ENABLE_TESTING "Build acceptance tests" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(PROTO_HEADER ${CMAKE_CURRENT_BINARY_DIR}/iq_stream.pb.h)
add_custom_command(OUTPUT ${PROTO_HEADER} ${CMAKE_CURRENT_BINARY_DIR}/iq_stream.pb.cc
    COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR}
        -I ${CMAKE_CURRENT_SOURCE_DIR}/proto -I ${Protobuf_INCLUDE_DIRS}
        ${CMAKE_CURRENT_SOURCE_DIR}/proto/iq_stream.proto
    DEPENDS proto/iq_stream.proto VERBATIM)
add_library(gnuradio-iqstream SHARED lib/protocol.cc lib/session.cc lib/transport.cc
    lib/blocks.cc lib/align_streams_impl.cc ${CMAKE_CURRENT_BINARY_DIR}/iq_stream.pb.cc)
add_library(gnuradio::iqstream ALIAS gnuradio-iqstream)
target_compile_definitions(gnuradio-iqstream PRIVATE gnuradio_iqstream_EXPORTS)
target_include_directories(gnuradio-iqstream PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/gnuradio/iqstream>
    PRIVATE lib)
target_link_libraries(gnuradio-iqstream PUBLIC gnuradio::gnuradio-runtime protobuf::libprotobuf
    PRIVATE gRPC::grpc++ Threads::Threads)
target_compile_options(gnuradio-iqstream PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(gnuradio-iqstream PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0
    EXPORT_NAME iqstream)
if(ENABLE_SANITIZERS)
    target_compile_options(gnuradio-iqstream PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(gnuradio-iqstream PUBLIC -fsanitize=address,undefined)
endif()
install(TARGETS gnuradio-iqstream EXPORT iqstreamTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/gnuradio/iqstream DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/gnuradio)
install(FILES ${PROTO_HEADER} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/gnuradio/iqstream)
install(FILES proto/iq_stream.proto proto/UPSTREAM proto/LICENSE.rustradio DESTINATION ${CMAKE_INSTALL_DATADIR}/gr-iqstream/proto)
set(CONFIG_DIR ${CMAKE_INSTALL_LIBDIR}/cmake/gnuradio-iqstream)
configure_package_config_file(cmake/gnuradio-iqstreamConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/gnuradio-iqstreamConfig.cmake INSTALL_DESTINATION ${CONFIG_DIR})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/gnuradio-iqstreamConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT iqstreamTargets NAMESPACE gnuradio:: DESTINATION ${CONFIG_DIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/gnuradio-iqstreamConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/gnuradio-iqstreamConfigVersion.cmake DESTINATION ${CONFIG_DIR})
if(ENABLE_PYTHON)
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    find_package(pybind11 2.12 CONFIG REQUIRED)
    pybind11_add_module(iqstream_python python/iqstream/bindings/module.cc)
    target_link_libraries(iqstream_python PRIVATE gnuradio-iqstream)
    set_target_properties(iqstream_python PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/python/gnuradio/iqstream)
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/python/gnuradio/iqstream)
    configure_file(python/iqstream/__init__.py ${CMAKE_CURRENT_BINARY_DIR}/python/gnuradio/iqstream/__init__.py COPYONLY)
    # Follow GNU Radio's install scheme, including Debian's dist-packages layout.
    set(IQSTREAM_PYTHON_DIR ${GR_PYTHON_DIR}/gnuradio/iqstream CACHE STRING "Python package installation directory")
    install(TARGETS iqstream_python LIBRARY DESTINATION ${IQSTREAM_PYTHON_DIR})
    install(FILES python/iqstream/__init__.py DESTINATION ${IQSTREAM_PYTHON_DIR})
    install(DIRECTORY grc/ DESTINATION ${CMAKE_INSTALL_DATADIR}/gnuradio/grc/blocks FILES_MATCHING PATTERN "*.yml")
endif()
if(ENABLE_TESTING)
    enable_testing()
    add_executable(qa_protocol tests/qa_protocol.cc)
    target_include_directories(qa_protocol PRIVATE lib)
    target_link_libraries(qa_protocol PRIVATE gnuradio-iqstream gRPC::grpc++)
    add_test(NAME protocol COMMAND qa_protocol)
    add_executable(qa_state tests/qa_state.cc)
    target_include_directories(qa_state PRIVATE lib)
    target_link_libraries(qa_state PRIVATE gnuradio-iqstream gRPC::grpc++)
    add_test(NAME state COMMAND qa_state)
    add_executable(qa_transport tests/qa_transport.cc)
    target_include_directories(qa_transport PRIVATE lib)
    target_link_libraries(qa_transport PRIVATE gnuradio-iqstream gRPC::grpc++)
    add_test(NAME transport COMMAND qa_transport)
    add_executable(qa_alignment tests/qa_alignment.cc)
    target_include_directories(qa_alignment PRIVATE lib)
    target_link_libraries(qa_alignment PRIVATE gnuradio-iqstream gnuradio::gnuradio-blocks)
    add_test(NAME alignment_cpp COMMAND qa_alignment)
    if(ENABLE_PYTHON)
        add_test(NAME flowgraphs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/qa_flowgraphs.py)
        add_test(NAME alignment COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/qa_alignment.py)
        set_tests_properties(flowgraphs PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python;LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH}")
        set_tests_properties(alignment PROPERTIES TIMEOUT 30
            ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python;LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH}")
    endif()
    set_tests_properties(protocol state transport alignment_cpp PROPERTIES TIMEOUT 30)
    set_tests_properties(protocol state transport alignment_cpp PROPERTIES
        ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH}")
    if(ENABLE_PYTHON)
        set_tests_properties(flowgraphs PROPERTIES TIMEOUT 90)
    endif()
endif()
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/gr-iqstream/examples)
install(FILES README.md COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/gr-iqstream)
